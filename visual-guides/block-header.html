import struct class BlockHeader: def __init__(self, raw_bytes): # 确保输入是 80 字节 if len(raw_bytes) != 80: raise ValueError("区块头必须是 80 字节") # 解析各个字段（小端序） self.version = struct.unpack(', raw_bytes[0:4])[0] self.prev_hash = raw_bytes[4:36][::-1] # 翻转成大端序显示 self.merkle_root = raw_bytes[36:68][::-1] self.timestamp = struct.unpack(', raw_bytes[68:72])[0] self.bits = raw_bytes[72:76] self.nonce = struct.unpack(', raw_bytes[76:80])[0] def __repr__(self): return f"BlockHeader(version={self.version}, timestamp={self.timestamp}, nonce={self.nonce})"
